function confettiBurst(){
  try{
    if(window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const cv=document.createElement('canvas'); cv.className='cfetti';
    cv.width=innerWidth; cv.height=innerHeight;
    document.body.appendChild(cv);
    const x=cv.getContext('2d'), cols=['#FFD700','#FF5A7A','#58DFA3','#5AA9FF','#FFC94D','#C77DFF'];
    const N=Math.min(140, Math.round(innerWidth/3)), P=[];
    for(let i=0;i<N;i++) P.push({ x:innerWidth*(0.15+0.7*(i/N)), y:-20-Math.round(60*(i%7)/7), vx:(((i*53)%100)/100-0.5)*5, vy:2+((i*29)%100)/28, r:4+((i*17)%6), c:cols[i%cols.length], a:((i*37)%360), va:(((i*7)%10)-5)*0.9 });
    let t0=null;
    const step=(ts)=>{ if(t0===null) t0=ts; const el=ts-t0;
      x.clearRect(0,0,cv.width,cv.height);
      P.forEach(p=>{ p.vy+=0.12; p.x+=p.vx; p.y+=p.vy; p.a+=p.va;
        x.save(); x.translate(p.x,p.y); x.rotate(p.a*Math.PI/180);
        x.globalAlpha=Math.max(0,1-el/1900); x.fillStyle=p.c; x.fillRect(-p.r/2,-p.r/2,p.r,p.r*1.5); x.restore(); });
      if(el<1900) requestAnimationFrame(step); else cv.remove();
    };
    requestAnimationFrame(step);
  }catch(e){}
}
